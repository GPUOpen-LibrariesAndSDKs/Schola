cmake_minimum_required(VERSION 3.20)
project(TagLearningTests LANGUAGES CXX)
add_executable(tag_learning_tests Main.cpp ../../Source/TagLearningRuntime/Private/TagLearner.cpp)
target_include_directories(tag_learning_tests PRIVATE ../../Source/TagLearningRuntime/Public)
target_include_directories(tag_learning_tests SYSTEM PRIVATE ../../ThirdParty/RLtools/include)
target_compile_features(tag_learning_tests PRIVATE cxx_std_17)
target_compile_definitions(tag_learning_tests PRIVATE NOMINMAX RL_TOOLS_DEBUG)
if(MSVC)
    string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
    target_compile_options(tag_learning_tests PRIVATE /O2 /GR- /EHs-c- /bigobj /wd4530 /fp:precise)
endif()
